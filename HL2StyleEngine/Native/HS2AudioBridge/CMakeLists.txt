cmake_minimum_required(VERSION 3.24)
project(HS2AudioBridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(SOLOUD "${CMAKE_CURRENT_LIST_DIR}/../vendor/soloud-e82fd32c1f62183922f08c14c814a02b58db1873")
file(GLOB CORE CONFIGURE_DEPENDS "${SOLOUD}/src/core/*.cpp")
add_library(HS2AudioBridge SHARED HS2AudioBridge.cpp ${CORE} "${SOLOUD}/src/backend/null/soloud_null.cpp")
target_include_directories(HS2AudioBridge PRIVATE "${SOLOUD}/include" "${SOLOUD}/src/backend/miniaudio")
target_compile_definitions(HS2AudioBridge PRIVATE WITH_NULL NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_link_libraries(HS2AudioBridge PRIVATE ole32 uuid winmm)
